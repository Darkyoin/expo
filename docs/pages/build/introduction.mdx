const countWorkingDays = (startDateStr, endDateStr, holidays = []) => {
  let count = 0;
  const start = new Date(startDateStr + "T00:00:00");
  const end   = new Date(endDateStr   + "T00:00:00");
  const holidaySet = new Set(holidays);
  const cur = new Date(start);
  while (cur <= end) {
    const ds = cur.toISOString().slice(0, 10);
    if (!holidaySet.has(ds)) count++;
    cur.setDate(cur.getDate() + 1);
  }
  return count;
};

const computeNetBetween = (txs, fromDate, toDate) => {
  let net = 0;
  txs.forEach((t) => {
    if (t.date < fromDate || t.date > toDate) return;
    if (t.kind === "income") net += t.amount;
    else if (t.kind === "expense") net -= t.amount;
  });
  return net;
};

// ── Donut Pie Chart ────────────────────────────────────────────────────────────
// Converts degrees to radians
const toRad = (deg) => (deg * Math.PI) / 180;

// Returns a point on a circle given center, radius and angle (0° = top)
const polarXY = (cx, cy, r, deg) => ({
  x: +(cx + r * Math.cos(toRad(deg - 90))).toFixed(5),
  y: +(cy + r * Math.sin(toRad(deg - 90))).toFixed(5),
});

/**
 * Builds an SVG path string for a donut slice.
 *
 * CRITICAL FIX: When a single slice represents 100% of the total, sweepDeg
 * would be 360°. An SVG arc from point A back to point A is degenerate and
 * renders nothing (or crashes). We handle this by splitting the full circle
 * into two 179.9999° arcs that together close the ring correctly.
 */
const donutSlicePath = (cx, cy, outerR, innerR, startDeg, sweepDeg) => {
  const sweep = Math.abs(sweepDeg);

  // Full-circle case: split into two half-arcs to avoid SVG degenerate arc
  if (sweep >= 359.9999) {
    const half = 179.9999;
    // First half-arc (outer: top → bottom, inner: bottom → top)
    const o1 = polarXY(cx, cy, outerR, startDeg);
    const o2 = polarXY(cx, cy, outerR, startDeg + half);
    const i1 = polarXY(cx, cy, innerR, startDeg + half);
    const i2 = polarXY(cx, cy, innerR, startDeg);
    // Second half-arc
    const o3 = polarXY(cx, cy, outerR, startDeg + half);
    const o4 = polarXY(cx, cy, outerR, startDeg + sweep);
    const i3 = polarXY(cx, cy, innerR, startDeg + sweep);
    const i4 = polarXY(cx, cy, innerR, startDeg + half);

    return [
      // First arc
      `M ${o1.x} ${o1.y}`,
      `A ${outerR} ${outerR} 0 0 1 ${o2.x} ${o2.y}`,
      `L ${i1.x} ${i1.y}`,
      `A ${innerR} ${innerR} 0 0 0 ${i2.x} ${i2.y}`,
      "Z",
      // Second arc
      `M ${o3.x} ${o3.y}`,
      `A ${outerR} ${outerR} 0 0 1 ${o4.x} ${o4.y}`,
      `L ${i3.x} ${i3.y}`,
      `A ${innerR} ${innerR} 0 0 0 ${i4.x} ${i4.y}`,
      "Z",
    ].join(" ");
  }

  // Normal partial arc
  const clampedSweep = Math.min(sweep, 359.9998);
  const endDeg   = startDeg + clampedSweep;
  const largeArc = clampedSweep > 180 ? 1 : 0;

  const o1 = polarXY(cx, cy, outerR, startDeg);
  const o2 = polarXY(cx, cy, outerR, endDeg);
  const i1 = polarXY(cx, cy, innerR, endDeg);
  const i2 = polarXY(cx, cy, innerR, startDeg);

  return [
    `M ${o1.x} ${o1.y}`,
    `A ${outerR} ${outerR} 0 ${largeArc} 1 ${o2.x} ${o2.y}`,
    `L ${i1.x} ${i1.y}`,
    `A ${innerR} ${innerR} 0 ${largeArc} 0 ${i2.x} ${i2.y}`,
    "Z",
  ].join(" ");
};

function DonutChart({ slices, size = 200 }) {
  // Filter out zero-value slices — they cause degenerate 0° arcs
  const validSlices = (slices || []).filter((s) => s.value > 0);

  if (!validSlices.length) {
    return (
      <View style={{ height: 160, alignItems: "center", justifyContent: "center" }}>
        <Text style={{ color: C.muted, fontSize: 13 }}>ไม่มีข้อมูล</Text>
      </View>
    );
  }

  const total  = validSlices.reduce((s, x) => s + x.value, 0);
  const cx     = size / 2;
  const cy     = size / 2;
  const outerR = size / 2 - 6;
  const innerR = outerR * 0.54;

  // Use a small gap between slices only when more than one slice exists
  const GAP_DEG = validSlices.length > 1 ? 1.2 : 0;

  // Build cumulative slice geometry
  let cursor = 0;
  const computed = validSlices.map((sl) => {
    const pct   = sl.value / total;
    const sweep = pct * 360;
    const start = cursor;
    cursor += sweep;
    return { ...sl, pct, sweep, startDeg: start };
  });

  return (
    <View style={{ alignItems: "center" }}>
      {/* ── SVG Donut ── */}
      <View style={{ width: size, height: size }}>
        <Svg width={size} height={size}>
          {/* Track ring — background circle */}
          <Circle
            cx={cx}
            cy={cy}
            r={(outerR + innerR) / 2}
            fill="none"
            stroke={C.border}
            strokeWidth={outerR - innerR}
          />

          {/* Slices */}
          <G>
            {computed.map((sl, i) => {
              // Apply gap by shrinking sweep and offsetting start
              const gapHalf = GAP_DEG / 2;
              const start   = sl.startDeg + gapHalf;
              const sweep   = sl.sweep - GAP_DEG;

              // Skip slivers that collapsed to nothing after gap removal
              if (sweep <= 0) return null;

              return (
                <Path
                  key={i}
                  d={donutSlicePath(cx, cy, outerR, innerR, start, sweep)}
                  fill={sl.color}
                />
              );
            })}
          </G>
        </Svg>

        {/* Center hole label — RN View layered over SVG */}
        <View
          pointerEvents="none"
          style={{
            position: "absolute",
            width:  innerR * 2,
            height: innerR * 2,
            borderRadius: innerR,
            backgroundColor: C.card,
            top:  cy - innerR,
            left: cx - innerR,
            alignItems: "center",
            justifyContent: "center",
          }}
        >
          <Text style={{ color: C.muted, fontSize: 10 }}>รวม</Text>
          <Text style={{ color: C.text, fontWeight: "800", fontSize: 13 }}>
            ฿{fmt(total)}
          </Text>
        </View>
      </View>

      {/* ── Legend list ── */}
      <View style={{ marginTop: 16, width: "100%" }}>
        {validSlices.map((sl, i) => (
          <View
            key={i}
            style={{ flexDirection: "row", alignItems: "center", gap: 8, marginBottom: 6 }}
          >
            <View
              style={{ width: 12, height: 12, borderRadius: 3, backgroundColor: sl.color }}
            />
            <Text
              style={{ color: C.mutedLight, fontSize: 12, flex: 1 }}
              numberOfLines={1}
            >
              {sl.label}
            </Text>
            <Text style={{ color: C.text, fontSize: 12, fontWeight: "600" }}>
              ฿{fmt(sl.value)}
            </Text>
            <Text style={{ color: C.muted, fontSize: 11, width: 36, textAlign: "right" }}>
              {Math.round((sl.value / total) * 100)}%
            </Text>
          </View>
        ))}
      </View>
    </View>
  );
}

// ── Ring Progress Chart ────────────────────────────────────────────────────────
function RingProgress({ pct, size = 120, color = C.orange, children }) {
  const strokeWidth = size * 0.12;
  const radius = (size - strokeWidth) / 2;
  const cx = size / 2;
  const cy = size / 2;
  const circumference = 2 * Math.PI * radius;

  const clampedPct = Math.min(Math.max(pct, 0), 100);
  const strokeDashoffset = circumference - (clampedPct / 100) * circumference;

  const overPct = Math.min(Math.max(pct - 100, 0), 100);
  const overDashoffset = circumference - (overPct / 100) * circumference;

  const over2Pct = Math.min(Math.max(pct - 200, 0), 100);
  const over2Dashoffset = circumference - (over2Pct / 100) * circumference;

  const baseColor =
    pct >= 200 ? C.cyan   :
    pct >= 150 ? C.purple :
    pct >= 100 ? C.gold   :
    color;

  const holeSize = size - strokeWidth * 2;

  return (
    <View style={{ width: size, height: size, alignItems: "center", justifyContent: "center" }}>
      <Svg
        width={size}
        height={size}
        style={{ position: "absolute", top: 0, left: 0 }}
      >
        {/* Rotate the entire group so arc starts at 12 o'clock */}
        <G rotation="-90" origin={`${cx}, ${cy}`}>

          {/* Track ring */}
          <Circle
            cx={cx}
            cy={cy}
            r={radius}
            fill="none"
            stroke={C.border}
            strokeWidth={strokeWidth}
          />

          {/* Primary arc: 0–100% */}
          {clampedPct > 0 && (
            <Circle
              cx={cx}
              cy={cy}
              r={radius}
              fill="none"
              stroke={baseColor}
              strokeWidth={strokeWidth}
              strokeDasharray={circumference}
              strokeDashoffset={strokeDashoffset}
              strokeLinecap="round"
            />
          )}

          {/* Overflow arc: 100–200% rendered in gold */}
          {pct > 100 && overPct > 0 && (
            <Circle
              cx={cx}
              cy={cy}
              r={radius}
              fill="none"
              stroke={C.gold}
              strokeWidth={strokeWidth * 0.55}
              strokeDasharray={circumference}
              strokeDashoffset={overDashoffset}
              strokeLinecap="round"
              opacity={0.9}
            />
          )}

          {/* Overflow arc: 200%+ rendered in cyan */}
          {pct > 200 && over2Pct > 0 && (
            <Circle
              cx={cx}
              cy={cy}
              r={radius}
              fill="none"
              stroke={C.cyan}
              strokeWidth={strokeWidth * 0.35}
              strokeDasharray={circumference}
              strokeDashoffset={over2Dashoffset}
              strokeLinecap="round"
              opacity={0.95}
            />
          )}

        </G>
      </Svg>

      {/* Centered hole content */}
      <View
        style={{
          width: holeSize,
          height: holeSize,
          borderRadius: holeSize / 2,
          backgroundColor: "#110c00",
          alignItems: "center",
          justifyContent: "center",
          zIndex: 2,
        }}
      >
        {children}
      </View>
    </View>
  );
}
